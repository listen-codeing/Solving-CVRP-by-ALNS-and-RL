**步骤1：扩展数据结构**
```python
# vrp_data_tw.py
class VRPTWData(VRPData):
    def __init__(self, points, vehicle_capacity, node_demand,
                 routes, cost, time_windows, service_times):
        super().__init__(points, vehicle_capacity, node_demand, routes, cost)
        self.time_windows = time_windows
        self.service_times = service_times
        self.speed = 1.0  # 假设速度为1

    def compute_arrival_times(self, route):
        """计算路线的到达时间"""
        times = [0]  # depot at time 0
        for i in range(len(route)-1):
            prev_node = route[i]
            next_node = route[i+1]
            travel_time = self.distance_matrix[prev_node, next_node] / self.speed
            service_time = self.service_times[prev_node]
            arrival = max(times[-1] + service_time + travel_time,
                         self.time_windows[next_node][0])  # 等待至时间窗开始
            times.append(arrival)
        return times

    def is_time_feasible(self, route):
        """检查路线是否满足时间窗约束"""
        times = self.compute_arrival_times(route)
        for i, node in enumerate(route):
            early, late = self.time_windows[node]
            if times[i] > late:  # 迟到
                return False
        return True
```

```python
class RepairUtilsTW(RepairUtils):
    def get_min_insertion_cost(self, state, node):
        """考虑时间窗的插入成本"""
        min_cost = float('inf')
        best_position = None

        for i, route in enumerate(state.routes):
            for pos in range(1, len(route)):
                # 1. 检查容量
                if not self._check_capacity(state, route, node):
                    continue

                # 2. 检查时间窗
                new_route = route[:pos] + [node] + route[pos:]
                if not state.is_time_feasible(new_route):
                    continue

                # 3. 计算成本
                cost = self.insertion_cost(state, node,
                                          route[pos-1], route[pos])
                if cost < min_cost:
                    min_cost = cost
                    best_position = (i, pos)

        return min_cost, best_position
```

### 3.2 扩展到PDVRP（取送货VRP）

#### 额外的挑战

**1. 优先级约束**：必须先访问取货点，再访问送货点
```python
class PDVRPData(VRPData):
    def __init__(self, ..., pickup_delivery_pairs):
        super().__init__(...)
        self.pd_pairs = pickup_delivery_pairs  # {pickup: delivery}

    def is_precedence_satisfied(self, route):
        """检查优先级约束"""
        visited = set()
        for node in route:
            if node in self.pd_pairs.values():  # 这是送货点
                pickup = self._get_pickup(node)
                if pickup not in visited:
                    return False
            visited.add(node)
        return True
```

**2. 配对销毁**：销毁时必须同时移除取送货对
```python
class PairedRemoval(Destroy):
    def action(self, state, num_pairs_to_remove):
        pairs_to_remove = random.sample(
            list(state.pd_pairs.items()),
            k=num_pairs_to_remove
        )
        customers_to_remove = []
        for pickup, delivery in pairs_to_remove:
            customers_to_remove.extend([pickup, delivery])
        return self.remove_customers(state, customers_to_remove)
```

**3. 配对修复**：修复时必须插入完整的取送货对
```python
class PairedRepair(RepairUtils):
    def insert_customers(self, state):
        while state.unassigned_customers:
            # 找到一对未分配的取送货
            pickup = state.unassigned_customers[0]
            delivery = state.pd_pairs[pickup]

            # 找最佳插入位置（确保优先级）
            best_cost, route_id, pickup_pos, delivery_pos = \
                self._find_best_paired_insertion(state, pickup, delivery)

            # 插入
            state.routes[route_id].insert(pickup_pos, pickup)
            state.routes[route_id].insert(delivery_pos+1, delivery)

            state.unassigned_customers.remove(pickup)
            state.unassigned_customers.remove(delivery)

        return state
```

```
    ↓
```

```python
# framework/operator_registry.py
class OperatorRegistry:
    _destroy_ops = {}
    _repair_ops = {}

    @classmethod
    def register_destroy(cls, name):
        def decorator(op_class):
            cls._destroy_ops[name] = op_class
            return op_class
        return decorator

    @classmethod
    def register_repair(cls, name):
        def decorator(op_class):
            cls._repair_ops[name] = op_class
            return op_class
        return decorator

    @classmethod
    def create_operators(cls, config):
        destroy_ops = [
            cls._destroy_ops[name](**params)
            for name, params in config['destroy'].items()
        ]
        repair_ops = [
            cls._repair_ops[name](**params)
            for name, params in config['repair'].items()
        ]
        return destroy_ops, repair_ops
```

```python
@OperatorRegistry.register_destroy('shaw')
class ShawRemoval(Destroy):
    def __init__(self, relatedness_weights):
        self.weights = relatedness_weights

    def action(self, state, num_customers_to_remove):
        # 翻译C++逻辑
        seed = random.choice(state.customers)
        relatedness = self._compute_relatedness(state, seed)
        customers_to_remove = self._select_by_relatedness(
            relatedness, num_customers_to_remove
        )
        return self.remove_customers(state, customers_to_remove)

    def _compute_relatedness(self, state, seed):
        """计算与seed的相关度"""
        scores = {}
        for c in state.customers:
            if c == seed:
                continue
            score = (
                self.weights['distance'] * state.distance_matrix[seed, c] +
                self.weights['demand'] * abs(state.demand[seed] - state.demand[c]) +
                self.weights['time'] * self._time_difference(state, seed, c)
            )
            scores[c] = score
        return scores
```

## 八、结论

这个代码是一个**优秀的研究原型**，展示了RL+ALNS的可行性，但**不是一个生产级框架**。

**如果你的目标是**：
- ✅ 学习RL+ALNS的原理 → 现有代码足够
- ✅ 快速实验几个新操作符 → 直接修改现有代码

**我的建议**：
1. **阶段0（必做）**：先运行原型，理解RL效果
2. **阶段1（可选）**：如果要长期使用，投入重构
3. **阶段2-3（视情况）**：逐步迁移C++操作符
4. **阶段4（建议）**：扩展到更多VRP变形

这是一个可行的技术路线，但需要脚踏实地，分阶段实施。
